   392→/* ────────────────────────────── Animated Counter ────────────────────────────── */
   393→
   394→function AnimatedCounter({ end, suffix, label }: { end: number; suffix: string; label: string }) {
   395→  const [count, setCount] = useState(0);
   396→  const [started, setStarted] = useState(false);
   397→  const ref = useRef<HTMLDivElement>(null);
   398→
   399→  useEffect(() => {
   400→    const el = ref.current;
   401→    if (!el) return;
   402→    const observer = new IntersectionObserver(
   403→      ([entry]) => { if (entry.isIntersecting && !started) setStarted(true); },
   404→      { threshold: 0.5 }
   405→    );
   406→    observer.observe(el);
   407→    return () => observer.disconnect();
   408→  }, [started]);
   409→
   410→  useEffect(() => {
   411→    if (!started) return;
   412→    let startTime: number;
   413→    const duration = 2000;
   414→    const animate = (timestamp: number) => {
   415→      if (!startTime) startTime = timestamp;
   416→      const progress = Math.min((timestamp - startTime) / duration, 1);
   417→      const eased = 1 - Math.pow(1 - progress, 3);
   418→      setCount(Math.floor(eased * end));
   419→      if (progress < 1) requestAnimationFrame(animate);
   420→    };
   421→    requestAnimationFrame(animate);
   422→  }, [started, end]);
   423→
   424→  return (
   425→    <div ref={ref} className="hero-stat-card px-5 py-4 text-center">
   426→      <div className="text-3xl md:text-4xl font-bold gradient-text mb-1">{count}{suffix}</div>
   427→      <div className="text-white/60 text-xs sm:text-sm font-medium">{label}</div>
   428→    </div>
   429→  );
   430→}
