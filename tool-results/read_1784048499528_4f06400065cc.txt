   365→   365→   365→
   366→   366→   366→function AnimatedCounter({ end, suffix, label }: { end: number; suffix: string; label: string }) {
   367→   367→   367→  const [count, setCount] = useState(0);
   368→   368→   368→  const [started, setStarted] = useState(false);
   369→   369→   369→  const ref = useRef<HTMLDivElement>(null);
   370→   370→   370→
   371→   371→   371→  useEffect(() => {
   372→   372→   372→    const el = ref.current;
   373→   373→   373→    if (!el) return;
   374→   374→   374→    const observer = new IntersectionObserver(
   375→   375→   375→      ([entry]) => { if (entry.isIntersecting && !started) setStarted(true); },
   376→   376→   376→      { threshold: 0.5 }
   377→   377→   377→    );
   378→   378→   378→    observer.observe(el);
   379→   379→   379→    return () => observer.disconnect();
   380→   380→   380→  }, [started]);
   381→   381→   381→
   382→   382→   382→  useEffect(() => {
   383→   383→   383→    if (!started) return;
   384→   384→   384→    let startTime: number;
   385→   385→   385→    const duration = 2000;
   386→   386→   386→    const animate = (timestamp: number) => {
   387→   387→   387→      if (!startTime) startTime = timestamp;
   388→   388→   388→      const progress = Math.min((timestamp - startTime) / duration, 1);
   389→   389→   389→      const eased = 1 - Math.pow(1 - progress, 3);
   390→   390→   390→      setCount(Math.floor(eased * end));
   391→   391→   391→      if (progress < 1) requestAnimationFrame(animate);
   392→   392→   392→    };
   393→   393→   393→    requestAnimationFrame(animate);
   394→   394→   394→  }, [started, end]);
   395→   395→   395→
   396→   396→   396→  return (
   397→   397→   397→    <div ref={ref} className="hero-stat-card px-5 py-4 text-center">
   398→   398→   398→      <div className="text-3xl md:text-4xl font-bold gradient-text mb-1">{count}{suffix}</div>
   399→   399→   399→      <div className="text-white/60 text-xs sm:text-sm font-medium">{label}</div>
   400→   400→   400→    </div>
   401→   401→   401→  );
   402→   402→   402→}
