   335→/* ────────────────────────────── Animated Counter Component ────────────────────────────── */
   336→
   337→function AnimatedCounter({ end, suffix, label }: { end: number; suffix: string; label: string }) {
   338→  const [count, setCount] = useState(0);
   339→  const [started, setStarted] = useState(false);
   340→  const ref = useRef<HTMLDivElement>(null);
   341→
   342→  useEffect(() => {
   343→    const el = ref.current;
   344→    if (!el) return;
   345→    const observer = new IntersectionObserver(
   346→      ([entry]) => {
   347→        if (entry.isIntersecting && !started) {
   348→          setStarted(true);
   349→        }
   350→      },
   351→      { threshold: 0.5 }
   352→    );
   353→    observer.observe(el);
   354→    return () => observer.disconnect();
   355→  }, [started]);
   356→
   357→  useEffect(() => {
   358→    if (!started) return;
   359→    let startTime: number;
   360→    const duration = 2000;
   361→    const animate = (timestamp: number) => {
   362→      if (!startTime) startTime = timestamp;
   363→      const progress = Math.min((timestamp - startTime) / duration, 1);
   364→      const eased = 1 - Math.pow(1 - progress, 3);
   365→      setCount(Math.floor(eased * end));
   366→      if (progress < 1) requestAnimationFrame(animate);
   367→    };
   368→    requestAnimationFrame(animate);
   369→  }, [started, end]);
   370→
   371→  return (
   372→    <div ref={ref} className="text-center">
   373→      <div className="text-3xl md:text-4xl font-bold gradient-text mb-1">{count}{suffix}</div>
   374→      <div className="text-white/50 text-xs sm:text-sm">{label}</div>
   375→    </div>
   376→  );
   377→}
